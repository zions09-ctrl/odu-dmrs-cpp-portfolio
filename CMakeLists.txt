cmake_minimum_required(VERSION 3.16)
project(odu_dmrs_cpp_portfolio VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(odu_dmrs STATIC src/controller.cpp)
target_include_directories(odu_dmrs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(odu_dmrs PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dmrs_demo apps/demo.cpp)
target_link_libraries(dmrs_demo PRIVATE odu_dmrs)
add_executable(dmrs_benchmark benchmarks/benchmark.cpp)
target_link_libraries(dmrs_benchmark PRIVATE odu_dmrs)
add_executable(dmrs_tests tests/test_controller.cpp)
target_link_libraries(dmrs_tests PRIVATE odu_dmrs Threads::Threads)
enable_testing()
add_test(NAME controller_tests COMMAND dmrs_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME historical_aggregates COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_historical_aggregates.py")
endif()
